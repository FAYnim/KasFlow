# Kasbon Feature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a cash advance (kasbon) recording feature with monthly filter to both public and admin views of the class cashflow system.

**Architecture:** New table `kasbon` (no FK), one public read-only API endpoint, five admin CRUD endpoints, new SPA tab in public app, new admin module. Pattern follows existing `jurnal_kas` and `mutasi_bank` implementations exactly.

**Tech Stack:** PHP native, PDO/MySQL, jQuery (existing), Tailwind CSS (CDN, existing).

```sql
-- Tabel Kasbon (Uang Dipinjamkan)
CREATE TABLE kasbon (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nama VARCHAR(100) NOT NULL,
    tanggal DATE NOT NULL,
    keterangan TEXT NOT NULL,
    jumlah DECIMAL(12,2) NOT NULL,
    status ENUM('belum_lunas','lunas') DEFAULT 'belum_lunas',
    tanggal_lunas DATE NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE INDEX idx_kasbon_tanggal ON kasbon (tanggal);
CREATE INDEX idx_kasbon_status ON kasbon (status);
```

```php
        case 'get_kasbon': {
            $bulanMap = ['Januari'=>1,'Februari'=>2,'Maret'=>3,'April'=>4,'Mei'=>5,'Juni'=>6,'Juli'=>7,'Agustus'=>8,'September'=>9,'Oktober'=>10,'November'=>11,'Desember'=>12];

            $where = []; $args = [];
            if (isset($bulanMap[$bulanIdx])) {
                $where[] = 'MONTH(tanggal) = ?';
                $args[]  = $bulanMap[$bulanIdx];
            } else {
                http_response_code(400);
                echo json_encode(['error' => 'bulan tidak valid']);
                break;
            }
            $where[] = 'YEAR(tanggal) = ?';
            $args[]  = $tahun;
            $sqlWhere = 'WHERE ' . implode(' AND ', $where);
            $stmt = $pdo->prepare("SELECT id, nama, tanggal, keterangan, jumlah, status, tanggal_lunas FROM kasbon $sqlWhere ORDER BY tanggal DESC, id DESC");
            $stmt->execute($args);
            echo json_encode($stmt->fetchAll());
            break;
        }
```

```php
        case 'add_kasbon': {
            $nama  = trim($_POST['nama'] ?? '');
            $tgl   = $_POST['tanggal'] ?? date('Y-m-d');
            $ket   = trim($_POST['keterangan'] ?? '');
            $jml   = (float)($_POST['jumlah'] ?? 0);
            $stat  = $_POST['status'] ?? 'belum_lunas';
            if ($nama === '' || $jml <= 0 || !in_array($stat, ['belum_lunas','lunas'], true)) {
                http_response_code(400); echo json_encode(['error' => 'invalid']); break;
            }
            $tLunas = ($stat === 'lunas') ? date('Y-m-d') : null;
            $pdo->prepare("INSERT INTO kasbon (nama, tanggal, keterangan, jumlah, status, tanggal_lunas) VALUES (?,?,?,?,?,?)")
                ->execute([$nama, $tgl, $ket, $jml, $stat, $tLunas]);
            echo json_encode(['ok' => true, 'id' => $pdo->lastInsertId()]);
            break;
        }
        case 'update_kasbon': {
            $id   = (int)($_POST['id'] ?? 0);
            $nama = trim($_POST['nama'] ?? '');
            $tgl  = $_POST['tanggal'] ?? date('Y-m-d');
            $ket  = trim($_POST['keterangan'] ?? '');
            $jml  = (float)($_POST['jumlah'] ?? 0);
            $stat = $_POST['status'] ?? 'belum_lunas';
            if ($id <= 0 || $nama === '' || $jml <= 0 || !in_array($stat, ['belum_lunas','lunas'], true)) {
                http_response_code(400); echo json_encode(['error' => 'invalid']); break;
            }
            $tLunas = ($stat === 'lunas') ? date('Y-m-d') : null;
            $pdo->prepare("UPDATE kasbon SET nama=?, tanggal=?, keterangan=?, jumlah=?, status=?, tanggal_lunas=? WHERE id=?")
                ->execute([$nama, $tgl, $ket, $jml, $stat, $tLunas, $id]);
            echo json_encode(['ok' => true]);
            break;
        }
        case 'mark_lunas_kasbon': {
            $id = (int)($_POST['id'] ?? 0);
            if ($id <= 0) { http_response_code(400); echo json_encode(['error' => 'invalid id']); break; }
            $pdo->prepare("UPDATE kasbon SET status='lunas', tanggal_lunas=? WHERE id=?")
                ->execute([date('Y-m-d'), $id]);
            echo json_encode(['ok' => true]);
            break;
        }
        case 'mark_belum_lunas_kasbon': {
            $id = (int)($_POST['id'] ?? 0);
            if ($id <= 0) { http_response_code(400); echo json_encode(['error' => 'invalid id']); break; }
            $pdo->prepare("UPDATE kasbon SET status='belum_lunas', tanggal_lunas=NULL WHERE id=?")
                ->execute([$id]);
            echo json_encode(['ok' => true]);
            break;
        }
        case 'delete_kasbon': {
            $id = (int)($_REQUEST['id'] ?? 0);
            if ($id <= 0) { http_response_code(400); echo json_encode(['error' => 'invalid id']); break; }
            $pdo->prepare("DELETE FROM kasbon WHERE id=?")->execute([$id]);
            echo json_encode(['ok' => true]);
            break;
        }
```

```javascript
// Kasbon tab
function loadKasbon() {
    const bulan = $('#kasbon-bulan').val();
    const tahun = $('#kasbon-tahun').val();
    $.getJSON('src/api/public.php', { action: 'get_kasbon', bulan: bulan, tahun: tahun }, function(data) {
        const tbody = $('#kasbon-table-body');
        tbody.empty();
        if (!data.length) {
            tbody.append('<tr><td colspan="6" class="text-center px-4 py-8 text-[#8a8f98]">Tidak ada data kasbon</td></tr>');
            return;
        }
        data.forEach(function(r, i) {
            const badge = r.status === 'lunas'
                ? '<span class="inline-block px-2 py-0.5 rounded-full text-xs font-medium bg-green-900/40 text-green-400">Lunas</span>'
                : '<span class="inline-block px-2 py-0.5 rounded-full text-xs font-medium bg-yellow-900/40 text-yellow-400">Belum Lunas</span>';
            tbody.append(
                '<tr class="border-b border-[#1a1b1e]">' +
                '<td class="px-4 py-3 text-[#8a8f98]">' + (i + 1) + '</td>' +
                '<td class="px-4 py-3">' + r.tanggal + '</td>' +
                '<td class="px-4 py-3">' + r.nama + '</td>' +
                '<td class="px-4 py-3 text-[#8a8f98]">' + r.keterangan + '</td>' +
                '<td class="px-4 py-3 text-right font-mono">' + formatRupiah(r.jumlah) + '</td>' +
                '<td class="px-4 py-3">' + badge + '</td>' +
                '</tr>'
            );
        });
    });
}

$(function() {
    // Populate tahun dropdown

    for (let y = thisYear; y >= thisYear - 2; y--) {

    $('#kasbon-bulan').val(monthNames[new Date().getMonth()]);
```

```javascript
// ---- Kasbon Admin ----
function loadAdminKasbon() {
    const bulan = $('#admin-kasbon-bulan').val();
    const tahun = $('#admin-kasbon-tahun').val();
    $.getJSON('../src/api/public.php', { action: 'get_kasbon', bulan: bulan, tahun: tahun }, function(data) {
        const tbody = $('#admin-kasbon-table-body');
        tbody.empty();
        if (!data.length) {
            tbody.append('<tr><td colspan="7" class="text-center px-4 py-8 text-[#8a8f98]">Tidak ada data kasbon</td></tr>');
            return;
        }
        data.forEach(function(r, i) {
            const badge = r.status === 'lunas'
                ? '<span class="inline-block px-2 py-0.5 rounded-full text-xs font-medium bg-green-900/40 text-green-400">Lunas</span>'
                : '<span class="inline-block px-2 py-0.5 rounded-full text-xs font-medium bg-yellow-900/40 text-yellow-400">Belum Lunas</span>';
            const toggleBtn = r.status === 'lunas'
                ? '<button onclick="kasbonToggleStatus(' + r.id + ', \'belum_lunas\')" class="text-yellow-400 hover:text-yellow-300 text-xs" title="Tandai Belum Lunas"><i class="fa-solid fa-rotate-left"></i></button>'
                : '<button onclick="kasbonToggleStatus(' + r.id + ', \'lunas\')" class="text-green-400 hover:text-green-300 text-xs" title="Tandai Lunas"><i class="fa-solid fa-circle-check"></i></button>';
            tbody.append(
                '<tr class="border-b border-[#1a1b1e]">' +
                '<td class="px-4 py-3 text-[#8a8f98]">' + (i + 1) + '</td>' +
                '<td class="px-4 py-3">' + r.tanggal + '</td>' +
                '<td class="px-4 py-3">' + r.nama + '</td>' +
                '<td class="px-4 py-3 text-[#8a8f98]">' + r.keterangan + '</td>' +
                '<td class="px-4 py-3 text-right font-mono">' + formatRupiah(r.jumlah) + '</td>' +
                '<td class="px-4 py-3">' + badge + '</td>' +
                '<td class="px-4 py-3 text-center">' +
                    '<button onclick="kasbonEdit(' + JSON.stringify(r).replace(/"/g, '&quot;') + ')" class="text-[#828fff] hover:text-[#5e6ad2] text-xs mr-2" title="Edit"><i class="fa-solid fa-pen-to-square"></i></button>' +
                    toggleBtn +
                    ' <button onclick="kasbonDelete(' + r.id + ')" class="text-red-400 hover:text-red-300 text-xs" title="Hapus"><i class="fa-solid fa-trash"></i></button>' +
                '</td>' +
                '</tr>'
            );
        });
    });
}

function kasbonEdit(r) {
    $('#kasbon-edit-id').val(r.id);
    $('#kasbon-nama').val(r.nama);
    $('#kasbon-tanggal').val(r.tanggal);
    $('#kasbon-keterangan').val(r.keterangan);
    $('#kasbon-jumlah').val(r.jumlah);
    $('#kasbon-status').val(r.status);
    $('#kasbon-submit-btn').text('Update');
    $('#kasbon-cancel-btn').removeClass('hidden');
}

function kasbonToggleStatus(id, newStatus) {
    const action = newStatus === 'lunas' ? 'mark_lunas_kasbon' : 'mark_belum_lunas_kasbon';
    $.post('../src/api/admin.php', { action: action, id: id }, function(res) {
        loadAdminKasbon();
    });
}

function kasbonDelete(id) {
    if (!confirm('Hapus kasbon ini?')) return;
    $.ajax({ url: '../src/api/admin.php', type: 'DELETE', data: { action: 'delete_kasbon', id: id }, success: function() { loadAdminKasbon(); } });
}

$(function() {
    const thisYear = new Date().getFullYear();
    const $yr = $('#admin-kasbon-tahun');
    for (let y = thisYear; y >= thisYear - 2; y--) {
        $yr.append('<option value="' + y + '">' + y + '</option>');
    }
    $yr.val(thisYear);
    const monthNames = ['Januari','Februari','Maret','April','Mei','Juni','Juli','Agustus','September','Oktober','November','Desember'];
    $('#admin-kasbon-bulan').val(monthNames[new Date().getMonth()]);

    $('#admin-kasbon-bulan, #admin-kasbon-tahun').on('change', loadAdminKasbon);

    // Form submit
    $('#form-kasbon').on('submit', function(e) {
        e.preventDefault();
        const editId = $('#kasbon-edit-id').val();
        const payload = {
            action: editId ? 'update_kasbon' : 'add_kasbon',
            id: editId || undefined,
            nama: $('#kasbon-nama').val(),
            tanggal: $('#kasbon-tanggal').val(),
            keterangan: $('#kasbon-keterangan').val(),
            jumlah: $('#kasbon-jumlah').val(),
            status: $('#kasbon-status').val()
        };
        $.post('../src/api/admin.php', payload, function(res) {
            if (res.ok) {
                $('#form-kasbon')[0].reset();
                $('#kasbon-edit-id').val('');
                $('#kasbon-submit-btn').text('Tambah');
                $('#kasbon-cancel-btn').addClass('hidden');
                loadAdminKasbon();
            }
        });
    });

    $('#kasbon-cancel-btn').on('click', function() {
        $('#form-kasbon')[0].reset();
        $('#kasbon-edit-id').val('');
        $('#kasbon-submit-btn').text('Tambah');
        $(this).addClass('hidden');
    });
});
```
